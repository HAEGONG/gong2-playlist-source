cmake_minimum_required(VERSION 3.28)

project(gong2-playlist-source VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(PLUGIN_NAME "gong2-playlist-source")

find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

add_library(${PLUGIN_NAME} MODULE)

target_sources(
  ${PLUGIN_NAME}
  PRIVATE
    src/plugin-main.cpp
    src/playlist-source.hpp
    src/playlist-source.cpp
    src/resume-store.hpp
    src/resume-store.cpp
    src/playlist-window.hpp
    src/playlist-window.cpp
)

target_compile_definitions(${PLUGIN_NAME} PRIVATE PLUGIN_VERSION="${PROJECT_VERSION}")
target_link_libraries(${PLUGIN_NAME} PRIVATE Qt6::Core Qt6::Widgets)
set_target_properties(${PLUGIN_NAME} PROPERTIES AUTOMOC ON PREFIX "")

if(APPLE)
  # Local macOS build: headers come from the OBS source tree, libraries from the
  # installed OBS.app, so the plugin is built against exactly the OBS you run.
  set(OBS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/obs-studio" CACHE PATH "obs-studio source tree (headers)")
  set(OBS_APP "/Applications/OBS.app" CACHE PATH "Installed OBS.app to link against")
  set(OBS_DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/obs-deps" CACHE PATH "obs-deps prebuilt package (for simde headers)")

  if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs-module.h")
    message(FATAL_ERROR "OBS_SOURCE_DIR does not contain libobs headers: ${OBS_SOURCE_DIR}")
  endif()

  # obsconfig.h is generated by the OBS build; plugins only need the macros to exist.
  set(OBS_DATA_PATH "../Resources")
  set(OBS_PLUGIN_PATH "../PlugIns")
  set(OBS_PLUGIN_DESTINATION "PlugIns")
  set(OBS_RELEASE_CANDIDATE 0)
  set(OBS_BETA 0)
  configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_CURRENT_BINARY_DIR}/obs-config/obsconfig.h")

  target_include_directories(
    ${PLUGIN_NAME}
    PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api" "${CMAKE_CURRENT_BINARY_DIR}/obs-config"
  )
  target_include_directories(${PLUGIN_NAME} SYSTEM PRIVATE "${OBS_DEPS_DIR}/include")
  target_link_options(${PLUGIN_NAME} PRIVATE "-F${OBS_APP}/Contents/Frameworks")
  target_link_libraries(
    ${PLUGIN_NAME}
    PRIVATE "-framework libobs" "${OBS_APP}/Contents/Frameworks/obs-frontend-api.dylib"
  )
  target_compile_options(${PLUGIN_NAME} PRIVATE -Wall -Wno-quoted-include-in-framework-header)

  set_target_properties(
    ${PLUGIN_NAME}
    PROPERTIES
      BUNDLE TRUE
      BUNDLE_EXTENSION plugin
      MACOSX_BUNDLE_GUI_IDENTIFIER "com.haegong.gong2-playlist-source"
      MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
      INSTALL_RPATH "@executable_path/../Frameworks"
      BUILD_WITH_INSTALL_RPATH TRUE
  )

  # Module data (locale files) lives in Contents/Resources of the bundle.
  add_custom_command(
    TARGET ${PLUGIN_NAME}
    POST_BUILD
    COMMAND
      "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/data"
      "$<TARGET_BUNDLE_CONTENT_DIR:${PLUGIN_NAME}>/Resources"
  )

  set(OBS_USER_PLUGIN_DIR "$ENV{HOME}/Library/Application Support/obs-studio/plugins")
  install(TARGETS ${PLUGIN_NAME} LIBRARY DESTINATION "${OBS_USER_PLUGIN_DIR}")
else()
  # Windows / Linux: use the CMake packages of an OBS development install.
  find_package(libobs REQUIRED)
  find_package(obs-frontend-api REQUIRED)
  target_link_libraries(${PLUGIN_NAME} PRIVATE OBS::libobs OBS::obs-frontend-api)

  if(WIN32)
    install(TARGETS ${PLUGIN_NAME} LIBRARY DESTINATION "obs-plugins/64bit")
    install(DIRECTORY data/ DESTINATION "data/obs-plugins/${PLUGIN_NAME}")
  else()
    include(GNUInstallDirs)
    install(TARGETS ${PLUGIN_NAME} LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/obs-plugins")
    install(DIRECTORY data/ DESTINATION "${CMAKE_INSTALL_DATAROOTDIR}/obs/obs-plugins/${PLUGIN_NAME}")
  endif()
endif()
